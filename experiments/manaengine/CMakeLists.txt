cmake_minimum_required(VERSION 3.21)
project(ManaEnginePrototype VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(PYBIND11_FINDPYTHON ON)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
add_library(manaengine_core src/engine.cpp src/damage_group.cpp src/pool_manifest.cpp)
target_include_directories(manaengine_core PUBLIC include)
target_compile_features(manaengine_core PUBLIC cxx_std_20)
set_target_properties(manaengine_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(manaengine_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(manaengine_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
pybind11_add_module(manaengine_native src/python_bindings.cpp)
target_link_libraries(manaengine_native PRIVATE manaengine_core)
set_target_properties(manaengine_native PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/python" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/python")
include(CTest)
if(BUILD_TESTING)
  add_executable(manaengine_tests tests/native_tests.cpp)
  target_link_libraries(manaengine_tests PRIVATE manaengine_core)
  add_test(NAME manaengine_native COMMAND manaengine_tests)
endif()
